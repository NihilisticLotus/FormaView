cmake_minimum_required(VERSION 3.25)
project(FormaView VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FORMAVIEW_BUILD_HOST "Build the Qt engineering host" ON)
option(FORMAVIEW_ASAN "Enable AddressSanitizer" OFF)
option(FORMAVIEW_UBSAN "Enable UndefinedBehaviorSanitizer (Clang/GCC)" OFF)
option(FORMAVIEW_QML_PREVIEW "Enable QML debug services in development configurations" OFF)

add_library(formaview_options INTERFACE)
if(MSVC)
    target_compile_options(formaview_options INTERFACE /W4 /WX /permissive- /Zc:__cplusplus /utf-8)
else()
    target_compile_options(formaview_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(FORMAVIEW_ASAN)
    if(MSVC)
        target_compile_options(formaview_options INTERFACE /fsanitize=address /Zi)
        target_link_options(formaview_options INTERFACE /INCREMENTAL:NO)
    else()
        target_compile_options(formaview_options INTERFACE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(formaview_options INTERFACE -fsanitize=address)
    endif()
endif()
if(FORMAVIEW_UBSAN)
    if(MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "MSVC does not provide UBSan; use Clang/GCC with the core-only build")
    endif()
    target_compile_options(formaview_options INTERFACE -fsanitize=undefined -fno-omit-frame-pointer)
    target_link_options(formaview_options INTERFACE -fsanitize=undefined)
endif()

find_package(Threads REQUIRED)
add_subdirectory(src/core)
include(cmake/SkeletalDependencies.cmake)
include(cmake/ProfessionalDependencies.cmake)
include(cmake/RasterDependencies.cmake)
include(cmake/ImageConversion.cmake)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(FORMAVIEW_BUILD_HOST)
    find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Quick GuiPrivate ShaderTools)
    qt_standard_project_setup(REQUIRES 6.12)
    add_subdirectory(src/render)
    add_subdirectory(src/host)
endif()
